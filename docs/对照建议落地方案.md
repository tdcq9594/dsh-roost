## A. `sessions.js` 并发安全

### A.1 要解决的三件事

现在 `sessions.js:29-35` 是 `writeFileSync(file, JSON.stringify(map))` 整文件覆盖，`load()` 的 `catch {}` 把「文件损坏」和「文件不存在」混为一谈。后果有两种：

1. **跨写者互盖**：面板「一键改写 / 一键回退」（`rewrite.js:363-384` 直写整份）与 DSH 进程的 `set()` 撞上时，后写的那个把前一个的结果抹掉。
2. **损坏静默**：`JSON.parse` 失败 → `map = {chats:{}}` → 下一次 `set()` 用空映射覆盖掉本来还能人工抢救的文件，用户表现是「所有人的上下文突然断了」，日志里一个字都没有。

### A.2 改法

只改 `src/sessions.js`，接口完全不变（`get/entry/set/remove/patch/getEntry/forgetSession/list/size/reload` 十个方法签名一个不动），因此 38 处调用方零改动。

```js
import { readFileSync, writeFileSync, renameSync, mkdirSync, copyFileSync, readdirSync, unlinkSync } from 'node:fs';
import { randomBytes } from 'node:crypto';
import { dirname, join } from 'node:path';

/** 墓碑存活时长：只用来挡住「写前合并」，不需要长期记忆 */
const TOMB_TTL_MS = 10 * 60 * 1000;

export function createSessions({ dataDir, log = () => {} }) {
  const file = join(dataDir, 'sessions.json');
  let map = { chats: {} };
  /** chatId → 删除时刻。仅防合并复活，不拦显式 set */
  const tombs = new Map();

  /** 三态：null=确实没有，undefined=损坏/读失败，object=正常 */
  function readDisk() {
    let text;
    try { text = readFileSync(file, 'utf8'); }
    catch (e) { return e.code === 'ENOENT' ? null : undefined; }
    try {
      const raw = JSON.parse(text);
      const ok = raw && typeof raw === 'object' && !Array.isArray(raw)
        && raw.chats && typeof raw.chats === 'object' && !Array.isArray(raw.chats);
      return ok ? raw : undefined;
    } catch { return undefined; }
  }

  function quarantine() {
    try {
      const bak = `${file}.corrupted.${Date.now()}`;
      copyFileSync(file, bak);
      /** 只留最近 3 份：反复启动遇到同一个坏文件时，别把目录堆满 */
      const olds = readdirSync(dirname(file))
        .filter((n) => n.startsWith('sessions.json.corrupted.'))
        .sort()
        .slice(0, -3);
      for (const n of olds) { try { unlinkSync(join(dirname(file), n)); } catch {} }
      log(`会话映射损坏（解析失败或结构不对），已备份到 ${bak}`);
    } catch (e) {
      log(`会话映射损坏，备份也失败：${e.message}`);
    }
  }

  function load() {
    const disk = readDisk();
    if (disk === undefined) { quarantine(); map = { chats: {} }; return; }
    map = disk ?? { chats: {} };
  }
  load();

  function save() {
    const now = Date.now();
    /** TTL 只是兜底：防写盘长期失败时 tombs 无限增长 */
    for (const [k, ts] of tombs) if (now - ts > TOMB_TTL_MS) tombs.delete(k);
    try {
      mkdirSync(dirname(file), { recursive: true });
      const disk = readDisk();
      /**
       * **保留 chats 之外的顶层字段**（扩展机制方案里规划了 `overrides[chatId]`，
       * 见 `docs/扩展机制方案.md:668`）。只写 `{chats}` 会把将来的字段静默抹掉。
       */
      const merged = { ...(disk && typeof disk === 'object' ? disk : {}), chats: {} };
      /** 磁盘打底：别的写者（一键改写/回退、手工编辑）新增的键不能被我们抹掉 */
      if (disk?.chats) for (const [k, v] of Object.entries(disk.chats)) merged.chats[k] = v;
      /** 内存覆盖：这些键我们是权威。注意空 sessionId 也是合法记录，不能用真值判断跳过 */
      for (const [k, v] of Object.entries(map.chats)) merged.chats[k] = v;
      /** 墓碑：显式删掉的键不许被磁盘上的旧版本复活 */
      for (const k of tombs.keys()) delete merged.chats[k];

      const tmp = `${file}.${process.pid}.${randomBytes(4).toString('hex')}.tmp`;
      writeFileSync(tmp, JSON.stringify(merged, null, 2), 'utf8');
      renameSync(tmp, file);
      /**
       * **落盘成功就清墓碑**：墓碑的语义是「还没写下去的删除意图」，写完即失效。
       * 不清的话，10 分钟内任何一次 save 都会把这个键再从磁盘删一遍——
       * 另一个写者刚合法重建的同名会话会被我们无声删掉。
       * 只有 catch 分支（写盘失败）才保留墓碑，留给下次重试。
       */
      tombs.clear();
      /**
       * **不把 merged 回写进 map**：`size` / `list()` 的语义保持「本进程管过的会话」
       * 不变（`host.js:5922` 的 `chats: sessions.size` 统计的就是这个数）。
       * 不会丢别人的键——每次 save 都以磁盘打底重新合并。
       */
    } catch (e) {
      log(`会话映射写盘失败：${e.message}`);
    }
  }

  function drop(chatId) {
    delete map.chats[chatId];
    tombs.set(chatId, Date.now());
  }

  return {
    /** 重置基线：外部整份改写后调用，墓碑一并清（这是权威重置，不是一次删除） */
    reload() { tombs.clear(); load(); },

    get(chatId, aliases = []) { /* 原样保留 */ },
    entry(chatId) { /* 原样保留 */ },

    set(chatId, sessionId, { preset = '', title = '', kind = '' } = {}) {
      /** 显式写入即撤销墓碑：「清上下文」之后必须能重新绑定 */
      tombs.delete(chatId);
      map.chats[chatId] = { sessionId: String(sessionId), preset, title, kind, at: Date.now() };
      save();
    },

    remove(chatId) { drop(chatId); save(); },

    patch(chatId, fields = {}) { /* 原样保留，命中时 save() */ },

    getEntry(chatId) { /* 原样保留 */ },

    forgetSession(sessionId) {
      let n = 0;
      for (const [chatId, e] of Object.entries(map.chats)) {
        if (e.sessionId === sessionId) { drop(chatId); n += 1; }
      }
      if (n) save();
      return n;
    },

    get size() { return Object.keys(map.chats).length; },
    list() { /* 原样保留 */ },
  };
}
```

### A.3 三条规则怎么落实（对照 §8-A）

| 规则 | 落点 |
|---|---|
| 空 `sessionId` 不是缺失 | 合并用 `Object.entries` 遍历、不用真值判断；`host.js:5452` / `:7286` 的 `set(chatId,'')` 条目原样保留 |
| `reload()` 重置基线 | `reload()` 清 `tombs` 再 `load()`；`save()` 每次重读磁盘打底、不把合并结果回写 `map`，所以 `size`/`list()` 语义不变 |
| 墓碑只防合并 | `set()` 第一行 `tombs.delete(chatId)`；`remove()`/`forgetSession()` 才写墓碑；**落盘成功后 `tombs.clear()`**（写完即失效，否则会持续误删别人重建的同名键） |
| 不碰别的顶层字段 | `merged = { ...disk, chats: {} }`，只重写 `chats`（为将来的 `overrides[chatId]` 留位） |

### A.4 验收测试（新增 `test/sessions-store.test.js`）

1. 原子写：`set()` 后文件是完整 JSON，目录里**没有** `*.tmp` 残留。
2. 写前合并：先手工往磁盘塞 `B`，再 `set(A)` → 落盘 `A` 与 `B` 都在。
3. 内存优先：磁盘 `A.sessionId='old'`、内存 `set(A,'new')` → 落盘是 `new`。
4. 墓碑生效：`remove(A)` 后手工把 `A` 写回磁盘 → 再 `set(B)` → 落盘 `A` 不在。
5. 墓碑不拦显式 set：`remove(A)` → `set(A,'x')` → 落盘 `A` 存在。
6. 空 sessionId 保留：`set(A,'')` → 落盘 `A.sessionId === ''`。
7. `reload` 清墓碑：`remove(A)` → 手工把 `A` 写回 → `reload()` → `set(B)` → 落盘 `A` **在**（重置是权威的）。
8. 损坏备份：文件写 `{bad` → `createSessions()` 后出现 `sessions.json.corrupted.*`，且 `log` 被调用一次、`size === 0`。
9. 结构不对：文件是 `[]` 或 `{"chats":[]}` → 同样备份，不抛。
10. 文件不存在：不备份、不告警、`size === 0`。
11. **`size` / `list()` 语义不变**：磁盘先放一个内存没碰过的 `X` → `size` 不计入它；但 `set(A)` 落盘后文件里 `X` 仍在（这条是防止「合并」悄悄改掉统计口径）。
12. 备份不堆积：连续三次用坏文件启动 → 目录里最多 3 份 `sessions.json.corrupted.*`。
13. **墓碑落盘即清**：`remove(A)`（写盘成功）→ 手工把 `A` 重建回磁盘 → `set(B)` → 落盘里 `A` **还在**（不该被墓碑二次删除）。
14. **保留其他顶层字段**：磁盘先是 `{"overrides":{"x":1},"chats":{}}` → `set(A)` → 落盘 `overrides` 仍在、`chats.A` 也在。

## B. `/ping` 拆两段耗时

`main.js:300-302` 现在是 `pong（已运行 N 秒）`。两个时间点都现成，**不用改任何底层**：
`message.js:292/341/461` 的 `receivedAt`（框架收到时刻）、`:465` 的 `at = Number(d.timestamp) || receivedAt`（QQ 事件时间戳）。

```js
core('ping', { desc: '测试机器人是否在线' }, async (msg) => {
  const up = Math.floor(runtime.uptimeMs / 1000);
  const now = Date.now();
  const at = Number(msg.at) || 0;
  const recv = Number(msg.receivedAt) || 0;
  /** 时间戳缺失/为 0 时只回原样，不输出伪数据 */
  if (!at || !recv) return msg.reply(`pong（已运行 ${up} 秒）`);
  const total = Math.max(0, now - at);
  const net = Math.max(0, recv - at);
  const proc = Math.max(0, now - recv);
  await msg.reply([`pong（已运行 ${up} 秒）`, `⏱ ${total}ms`, `  ├ 网络 ${net}ms`, `  └ 处理 ${proc}ms`].join('\n'));
});
```

**说明**：这是**单向时间戳差**，不是 RTT（和官方 openclaw 的 `/bot-ping` 同口径）。`msg.at` 回退到 `receivedAt` 时 `net === 0`，属于正常退化。

## C. magic bytes 判断顺序反转

现状（`roost-payload.js:70-79`）：先信 `content-type`，命中 `OK_MIME` 直接返回，嗅头部只兜底。QQ 网关统一标 `image/png`，所以**兜底永不触发**，实际等于全信声明。

```js
function guessMime(buf, declared = '') {
  /** 先看文件头：QQ 网关把所有图片统一标 image/png，声明不可信 */
  if (buf && buf.length >= 12) {
    if (buf[0] === 0xff && buf[1] === 0xd8) return 'image/jpeg';
    if (buf[0] === 0x89 && buf.toString('latin1', 1, 4) === 'PNG') return 'image/png';
    if (buf.toString('latin1', 0, 3) === 'GIF') return 'image/gif';
    if (buf.toString('latin1', 0, 4) === 'RIFF' && buf.toString('latin1', 8, 12) === 'WEBP') return 'image/webp';
  }
  /** 太短 / 认不出才退回声明（别把正常的极小图误杀） */
  const d = String(declared).split(';')[0].trim().toLowerCase();
  return OK_MIME.test(d) ? d : '';
}
```

**行为变化**：声明 `image/png` 但实际是 JPEG 的图，以前落成 `.png`、`mediaType` 也错；现在按真实格式落盘。这是修 bug，不是改语义。
**回归**：`test/*vision*` 与图片相关用例；手工验「群里发一张 GIF，看模型能不能读到」。

## D. 引用正文上限 500 → 2000

两处，口径必须一致：

1. `roost-payload.js:194`（`buildRendered`，现役生效点）
2. `dsh-roost/inbound.js:45`（无 `rendered` 时的兜底）

做法：在 **`inbound.js` 顶部**定义 `export const QUOTE_MAX_CHARS = 2000;`，`inbound.js:45` 用常量；`roost-payload.js` 从 `./inbound.js` import 并在 `:194` 用它。

**为什么定义在 `inbound.js` 而不是 `roost-payload.js`**：`inbound.js` 现在是**零 import 的纯函数模块**（全文没有一条 `import`），而 `roost-payload.js` 已经依赖 `fs/os/path` 与 `roost-merge.js`。反过来定义会把 `fs/os/path` 拖进 `inbound.js` 的依赖链，虽然不成环（`inbound.js` 不引用 `roost-payload.js`），但没必要。方向定成 `roost-payload.js → inbound.js`，和 `journal-bridge.js` 引用 `inbound.js` 一致。

保留尾部省略号，不彻底取消：`slice(0, QUOTE_MAX_CHARS) + (超长 ? '…' : '')`。

**为什么不是"全放开"**：openclaw 是全文注入，但那是它自己的取舍；我们每轮 token 成本直接翻倍不划算，2000 字足够覆盖绝大多数引用。

**注意**：`ai-chat/inbound.js:135` 有一模一样的 500，但它已不在链路上，**不动**（改了也没效果，反而制造「两处不一致」的错觉）。

## E. 审批按钮桥（三步，必须按序）

### E1. WS 模式下补交互回执（前置，独立可上线）

**问题**：`putInteraction` 全仓唯一调用点是 `webhook.js:470`（HTTP 分支）。两个实例都跑 `websocket`，所以回调型按钮点了之后平台收不到 PUT。

**改法**：`gateway.js` 构造时已持有 `api`（`:46-47`），在 `#handle` 的 `DISPATCH` 分支、seq 去重之后、`onDispatch` 之前插入：

```js
/** 交互事件必须回执，否则客户端一直转圈。webhook 分支在 webhook.js:468-475，这里补 WS */
if (t === 'INTERACTION_CREATE') {
  const id = p.d?.id;
  const tt = Number(p.d?.type ?? 0);
  /** 只有 11/12 需要回；13/14/16/18/19 官方明确不用（同 webhook.js:463-465 的结论） */
  if (id && (tt === 11 || tt === 12)) {
    try {
      /** api.js:1093 的 putInteraction 返回 Promise，fire-and-forget，别 await 阻塞分发 */
      const r = this.api?.putInteraction?.(id, 0);
      if (r && typeof r.catch === 'function') r.catch(() => {});
    } catch { /* 回执失败不影响分发 */ }
  }
}
```

**放在去重之后**的理由：resume 重放时同一 `interaction_id` 不会重复 PUT（重复回执会被判非法）。
**这一步不改变任何现有行为**：现有按钮是 `type:2` 指令型，根本不产生 interaction。

### E2. notice 通道放行 interaction（前置）

`host.js:3010-3017` 现在只处理 `/AUDIT/i`，其余丢弃。改成：

```js
function deliverNoticeFromRoost(data) {
```

`onRoostInteraction` 在 E2 阶段**就是个空处理器**（只 `log.debug` 记下事件 type 与 `resolved` 的键名），因为此时既没有发送方、也没有 pending 表。E3 接上 pending 后再往里填逻辑。**所以 E2 是纯加法，不影响任何现有链路。**

### E3. 回调型按钮 + pending 绑定

1. `outbound-actions.js:152-156` 的 `take()` 列表要加一行，与现有标记同级：
